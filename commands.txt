<VirtualHost *:80>

  ServerName ncis.konza.go.ke
  ServerAdmin kkiprop@konza.go.ke

  # ----------------------------------------------------------------
  # Frontend
  # ----------------------------------------------------------------
  DocumentRoot /var/www/html/national-coffee-information-system-frontend/dist/national-coffee-information-system-frontend/browser
  <Directory "/var/www/html/national-coffee-information-system-frontend/dist/national-coffee-information-system-frontend/browser">
      AllowOverride All
      Require all granted
  </Directory>

  # ----------------------------------------------------------------
  # Backend(/api)
  # ----------------------------------------------------------------
  Alias /api /var/www/html/national-coffee-information-system-backend/public
  <Directory "/var/www/html/national-coffee-information-system-backend/public">
      AllowOverride All
      Require all granted
  </Directory>


  #ErrorLog ${APACHE_LOG_DIR}/ncis_error.log
  #CustomLog ${APACHE_LOG_DIR}/ncis_access.log combined

  RewriteEngine on
  RewriteCond %{SERVER_NAME} =ncis.konza.go.ke
  RewriteRule ^ https://%{SERVER_NAME}%{REQUEST_URI} [END,NE,R=permanent]
</VirtualHost>

$ a2ensite ncis.conf

systemctl reload apache2

=================================================supervisor=================================================
1. Install and update supervisor in Linux Server
sudo apt-get install supervisor
sudo vi /etc/supervisor/conf.d/ncis.conf
[program:ncis]
process_name=%(program_name)s_%(process_num)02d
command=/usr/bin/php /var/www/html/national-coffee-information-system-backend/artisan queue:work database --sleep=3 --tries=3
autostart=true
autorestart=true
user=root
numprocs=5
redirect_stderr=true
stdout_logfile=/var/www/html/national-coffee-information-system-backend/storage/logs/worker.log

2. Update
sudo supervisorctl reread
sudo supervisorctl update
sudo supervisorctl start 'ncis:*'

sudo supervisorctl start ncis.conf
supervisorctl restart ncis:*

import { Component, AfterViewInit, OnDestroy } from '@angular/core';
import * as L from 'leaflet';

@Component({
  selector: 'app-parcel-map',
  templateUrl: './parcel-map.component.html',
  styleUrls: ['./parcel-map.component.scss']
})
export class ParcelMapComponent implements AfterViewInit, OnDestroy {

  geoJsonData: any;

  private map!: L.Map;
  private parcelLayer?: L.GeoJSON;

  ngAfterViewInit(): void {
    this.initializeMap();
    this.loadGeoJSON();
  }

  private initializeMap(): void {
    this.map = L.map('map').setView([-0.13, 35.46], 13);

    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '&copy; OpenStreetMap contributors',
      maxZoom: 19
    }).addTo(this.map);
  }

  private loadGeoJSON(): void {
    // Replace this with your existing geoJsonData assignment/API call.
    fetch('assets/data/greater_than_4ha_joined.geojson')
      .then(response => {
        if (!response.ok) {
          throw new Error(`HTTP error: ${response.status}`);
        }
        return response.json();
      })
      .then(data => {
        this.geoJsonData = data;

        this.parcelLayer = L.geoJSON(data, {
          style: () => ({
            color: '#224326',
            weight: 2,
            fillColor: '#62a66b',
            fillOpacity: 0.45
          }),
          onEachFeature: (feature, layer) => {
            layer.bindPopup(
              `<pre>${JSON.stringify(feature.properties ?? {}, null, 2)}</pre>`
            );
          }
        }).addTo(this.map);

        const bounds = this.parcelLayer.getBounds();

        if (bounds.isValid()) {
          this.map.fitBounds(bounds, { padding: [20, 20] });
        }

        setTimeout(() => this.map.invalidateSize(), 100);
      })
      .catch(error => {
        console.error('GeoJSON loading failed:', error);
      });
  }

  ngOnDestroy(): void {
    this.map?.remove();
  }
}
