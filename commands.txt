"styles": [
      "node_modules/sweetalert2/src/sweetalert2.scss"
      "node_modules/sweetalert2/dist/sweetalert2.css"
  ],
"scripts": [
      "node_modules/sweetalert2/dist/sweetalert2.js"
  ],
2. Import in Component
import swal from 'sweetalert2';
3. Use in Component
swal({
  title: 'Error!',
  text: 'Do you want to continue',
  type: 'error',
  confirmButtonText: 'Cool'
})

===================================Toastr2===================================
1. Install
npm install @iqx-limited/ngx-toastr@2.0.0 --save
npm install @iqx-limited/ngx-toastr@^2.0.0 --save

2. Add angular.json
"styles": [
    "src/styles.scss",
    "node_modules/@iqx-limited/ngx-toastr/toastr.css",
      "./node_modules/@iqx-limited/ngx-toastr/toastr.scss"

  ],

2. Import in Component
import { provideAnimations } from '@angular/platform-browser/animations';
import { provideToastr } from '@iqx-limited/ngx-toastr';

3. app.config.ts:
export const appConfig: ApplicationConfig = {
  providers: [
    provideAnimations(), // Required for toast animations
    provideToastr({
      timeOut: 5000,
      positionClass: 'toast-top-right',
      preventDuplicates: true,
    }),
  ],
};

===================================Deployment===================================
J. Deploy Angular App to Apache Server
1. Install Apache2:
-Install from here: http://httpd.apache.org/docs/current/install.htm

2. Copy the dist/ folder in /var/www/ folder.

3. Create Apache configuration file for angular app:
$ sudo vim /etc/apache2/sites-available/angular_konza_oss_permitting_management_system.conf
<VirtualHost *:80>
  ServerName permitting-management-system.oss.konza.go.ke
  ServerAdmin kkiprop@konza.go.ke
  DocumentRoot /var/www/html/permitting-management-system-frontend/dist/permitting-management-system-frontend/browser
  <Directory “/var/www/html/permitting-management-system-frontend/dist/permitting-management-system-frontend/browser”>
    AllowOverride All
    Require all granted
  </Directory>
</VirtualHost>

4. Enable angular.conf file using the following command:
$ a2ensite angular_konza_oss_permitting_management_system.conf

sudo chown -R username:groupname /path/to/angular/project

4. Permissions
chmod -R 755 /var/www/html/permitting-management-system-frontend

5. Restart the apache service:
sudo systemctl restart apache2
sudo systemctl reload apache2

===================================CKEditor===================================
1. Installation
npm install ckeditor5 @ckeditor/ckeditor5-angular --legacy-peer-deps

2. Import the CKEditor in:

a.styles.scss
  "styles": [
    "node_modules/sweetalert2/src/sweetalert2.scss",
    "node_modules/sweetalert2/dist/sweetalert2.css",
    "src/scss/styles.scss",
    "node_modules/@iqx-limited/ngx-toastr/toastr.scss",
    "ckeditor5/ckeditor5.css"
  ],

OR

b. CSS globally (in src/styles.css or styles.scss):
@import "ckeditor5/ckeditor5.css";

3. Update
// src/app/app.component.ts
import { Component, ViewEncapsulation } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { CKEditorModule } from '@ckeditor/ckeditor5-angular';
import {
  ClassicEditor,
  Bold,
  Essentials,
  Heading,
  Indent,
  IndentBlock,
  Italic,
  Link,
  List,
  Paragraph,
  Table,
  Undo,
} from 'ckeditor5';

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [CKEditorModule, FormsModule],
  templateUrl: './app.component.html',
  styleUrls: ['./app.component.css'],
  encapsulation: ViewEncapsulation.None,   // ← required for CKEditor styles
})
export class AppComponent {
  public Editor = ClassicEditor;

  public editorData = '<p>Hello from <strong>CKEditor 5</strong>!</p>';

  public config = {
    licenseKey: 'GPL',   // use 'GPL' for the open-source version
    plugins: [
      Bold, Essentials, Heading,
      Indent, IndentBlock, Italic,
      Link, List, Paragraph, Table, Undo,
    ],
    toolbar: [
      'undo', 'redo', '|',
      'heading', '|',
      'bold', 'italic', '|',
      'link', 'insertTable', '|',
      'bulletedList', 'numberedList',
      'indent', 'outdent',
    ],
  };
}

.html
<!-- src/app/app.component.html -->
<h2>CKEditor 5 in Angular 21</h2>

<ckeditor
  [editor]="Editor"
  [config]="config"
  [(ngModel)]="editorData"
>
</ckeditor>

<hr />
<h3>Output (HTML):</h3>
<pre>{{ editorData }}</pre>

5. Optional: set editor height
In app.component.css:
css/* Target the editable area */
.ck-editor__editable_inline {
  min-height: 300px;
}
